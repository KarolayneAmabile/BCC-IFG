h2=np.array([[-1,0,1],
            [-2,0,2],
            [-1,0,1]])
h1=np.array([[-1,-2,-1],
        [0,0,0],
         [1,2,1]])

import matplotlib.pyplot as plt

#mediana
mediana=filters.median(img)
#Suavização
kernel= np.ones((3,3))/9
img_conv= convolve2d(img,kernel,mode='same',boundary='symm')
#unsharp mask
img_conv2= convolve2d(imgborrada,kernel,mode='same',boundary='symm')
mask=imgborrada-img_conv2
img_final=imgborrada+ 5*mask
img_final=np.clip(img_final,0,1)
print (mask.min(),mask.max())
#sobel
sobelx=convolve2d(imgborrada,h1,mode='same',boundary='symm')

import numpy as np

def butterworth(shape,d0,n):
  H=np.zeros(shape)
  for u in range(shape[0]):
    for v in range(shape[1]):
      D_uv=np.sqrt( (u-shape[0]/2)**2 + (v-shape[1]/2)**2 )
      H[u,v] = 1/ (1+ (D_uv/d0 )**(2*n))
  return H

def butterworth_band_reject(shape,d0,w,n):
  H=np.zeros(shape)
  for u in range(shape[0]):
    for v in range(shape[1]):
      D_uv=np.sqrt( (u-shape[0]/2)**2 + (v-shape[1]/2)**2 )
      if D_uv==0:
        H[u,v]=1
      else:
        H[u,v] = 1/ (1+ ((D_uv*w)/(D_uv**2-d0**2))**(2*n))
  return H
